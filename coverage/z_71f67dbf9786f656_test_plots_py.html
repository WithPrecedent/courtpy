15def make_cases() -> courtpy.Cases: 

16 """Returns cases of three courts, with what the plots draw.""" 

17 data = pd.DataFrame({ 

18 'court_name': ['tenth_circuit'] * 4 + ['fifth_circuit'] * 3 + ['dc_circuit'], 

19 'court_num': [10, 10, 10, 10, 5, 5, 5, 12], 

20 'lower_court_name': [ 

21 'district_of_colorado', 'district_of_colorado', 'district_of_kansas', None, 

22 'northern_district_of_texas', 'northern_district_of_texas', 

23 'southern_district_of_texas', 'district_of_columbia'], 

24 'year': [2008, 2008, 2009, 2010, 2008, 2009, 2009, 2010], 

25 'outcome_reversal': pd.array([True, False, False, True, False, False, True, pd.NA], dtype = 'boolean'), 

26 'criminal_firearm': [True, True, False, False, True, False, False, False], 

27 'type_immigration': [False, False, True, False, False, True, True, False], 

28 'type_criminal': [True, True, False, False, True, False, False, True], 

29 'judge_party1': [-1, -1, 1, 1, 1, 1, -1, 1], 

30 'judge_party2': [-1, 1, 1, 1, 1, -1, -1, 1], 

31 'judge_party3': [-1, 1, 1, None, 1, -1, 1, 1]}, 

32 index = pd.Index([f'c{n}' for n in range(8)], name = 'case_id')) 

33 return courtpy.Cases.create(data)

36def draw(name: str, item: amos.Dataset, **parameters: object) -> object: 

37 """Applies a plot from the library and returns its figure.""" 

38 amos.library.borrow(name)().apply(item, **parameters) 

39 return item.figures[name]

50def test_court_rates_are_measured_rates() -> None: 

51 figure = draw('court_rates', make_cases()) 

52 axes = figure.axes[0] 

53 # The courts are in the order of the table of courts, with their names. 

54 assert [t.get_text() for t in axes.get_yticklabels()] == ['Fifth Circuit', 'Tenth Circuit'] 

55 assert [round(b.get_width(), 4) for b in axes.patches] == [round(1 / 3, 4), 0.5] 

56 labels = [t.get_text() for t in axes.texts] 

57 assert labels == [' 33% (n = 3)', ' 50% (n = 4)'] 

58 # The rate of every court is a line, and there are no error bars. 

59 assert axes.lines[0].get_xdata()[0] == pytest.approx(3 / 7) 

60 assert not axes.collections

63def test_cases_over_time_has_a_panel_for_each_court() -> None: 

64 figure = draw('cases_over_time', make_cases()) 

65 shown = [a for a in figure.axes if a.get_visible()] 

66 assert [a.get_title() for a in shown] == ['Fifth Circuit', 'Tenth Circuit', 'D.C. Circuit'] 

67 tenth = shown[1].lines[0]

72def test_issues_over_time_counts_each_issue() -> None: 

73 figure = draw('issues_over_time', make_cases(), issues = 'criminal_firearm, type_immigration') 

74 fifth = next(a for a in figure.axes if a.get_title() == 'Fifth Circuit')

113def test_judge_network() -> None:

116 'case_id': ['a', 'a', 'a', 'b', 'b', 'b'],
