17def apply(name: str, **sections: str) -> dict[str, Any]: 

18 """Applies a rulebook to the sections of one case. 

19 

20 Args: 

21 name: the name of the rulebook. 

22 **sections: the sections of text, by name. 

23 

24 Returns: 

25 The columns that the rules made, with `None` for missing values. 

26 

27 """ 

28 table = pd.DataFrame({k: [v] for k, v in sections.items()}, dtype = 'str') 

29 _, made = parsers._apply(amos.parsers.Rulebook.create(name), table) 

30 return { 

31 k: None if pd.isna(v) else v for k, v in made.iloc[0].to_dict().items()}

56def test_register_rulebooks(tmp_path: pathlib.Path) -> None: 

57 # A study's folders make a rulebook of each topic, applied in the order 

58 # of the folders, and one that combines them. 

59 general, texas = tmp_path / 'general', tmp_path / 'texas' 

60 general.mkdir() 

61 texas.mkdir() 

62 header = 'column,name,kind,pattern\n' 

63 (general / 'oil.csv').write_text(header + 'opinion,oil_well,flags,well\n', encoding = 'utf-8') 

64 (texas / 'oil.csv').write_text(header + 'opinion,oil_rrc,flags,Railroad Commission\n', encoding = 'utf-8') 

65 (texas / 'cattle.csv').write_text(header + 'opinion,cattle_brand,flags,brand\n', encoding = 'utf-8') 

66 books = rules.register_rulebooks(general, texas, combined = 'texas_all', prefix = 'texas') 

67 assert list(books) == ['texas_cattle', 'texas_oil', 'texas_all'] 

68 assert books['texas_oil'].__name__ == 'ParseTexasOil' 

69 assert amos.parsers.Rulebook.create('texas_oil').names == ['oil_well', 'oil_rrc'] 

70 assert amos.parsers.Rulebook.create('parse_texas_all').names == [ 

71 'cattle_brand', 'oil_well', 'oil_rrc'] 

72 assert amos.library.classify('parse_texas_cattle') == 'parser' 

73 assert list(rules.register_rulebooks(texas, topics = ['oil'])) == ['oil'] 

74 with pytest.raises(FileNotFoundError, match = 'sheep'): 

75 rules.register_rulebooks(general, topics = ['sheep']) 

76 with pytest.raises(FileNotFoundError, match = 'no files of rules'): 

77 rules.register_rulebooks(tmp_path / 'empty')

107def read_rows(path: Any) -> list[dict[str, str]]: 

108 """Returns the rows of a CSV file of rules.

147 federal = amos.parsers.Rulebook.create('federal') 

148 lexis = amos.parsers.Rulebook.create('lexis_nexis') 

149 general_header = amos.parsers.Rulebook.create('header')

151 courtlistener_header = amos.parsers.Rulebook.create('courtlistener_header') 

152 assert len(federal) > 300 

153 for variable in ( 

154 'opinion_reverse', 'criminal_firearm', 

155 'civil_titlevii', 'constitutional_amend4', 'citations_case', 

156 'citations_statute', 'agencies', 'review_district'): 

157 assert variable in federal.names 

158 # The header rules are applied by source, not with the federal rules. 

159 assert 'party1' not in federal.names and 'panel_judges' not in federal.names

161 'party1', 'party2', 'panel_judges', 'authors', 'concurring', 

162 'dissenting', 'disposition_reverse', 'cite_federal_reporter', 'party1_appellant',

164 'party1_agencies', 'party2_agencies', 'history_agencies', 

165 'lower_court_num', 'lower_court_name') 

166 for book in (general_header, lexis_header, courtlistener_header): 

167 for variable in shared: 

168 assert variable in book.names 

169 for variable in ('court_num', 'court_name', 'notice_unpublished', 'future_grant', 'cite_lexis'): 

170 assert variable in lexis_header.names 

171 assert variable not in courtlistener_header.names 

172 assert 'caption' in courtlistener_header.names and 'caption' not in lexis_header.names

177 'judges', 'opinion_by', 'concurring_lines', 'dissenting_lines']
