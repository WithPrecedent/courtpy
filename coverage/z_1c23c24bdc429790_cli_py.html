1"""Command line interface for courtpy. 

2 

3Run `courtpy --help` (or `courtpy {command} --help`) in a terminal for help. 

4 

5| Command | Does | 

6| --- | --- | 

7| `courtpy key set` | Stores your CourtListener API key outside of any project. | 

8| `courtpy key show` | Shows where your key is found (with most of it hidden). | 

9| `courtpy key delete` | Removes your stored key. | 

10| `courtpy download` | Downloads cases from CourtListener's bulk data (no key needed), or with its API ("--method api", a key is needed). | 

11| `courtpy split` | Divides Lexis-Nexis files of many cases into one file per case. | 

12| `courtpy parse` | Parses saved cases into a table (a CSV or parquet file). | 

13| `courtpy rules` | Checks a rulebook and lists the variables it makes. | 

14| `courtpy judges` | Makes and saves the roster of judges from the Federal Judicial Center's files. | 

15| `courtpy run` | Collects, parses, and analyzes cases as a settings file describes. | 

16 

17Each command is a private function that takes the parsed arguments and 

18returns the program's exit code (0 for success). `_parser` connects each 

19command's name to its function. 

20 

21A study built on courtpy can have a program of its own with courtpy's 

22commands and its own, by passing a function that adds them to `main`: 

23 

24```python 

25def main(argv: Sequence[str] | None = None) -> int: 

26 return courtpy.cli.main( 

27 argv, prog = 'my-study', add_commands = _add_commands) 

28 

29 

30def _add_commands(commands: Any) -> None: 

31 scores = commands.add_parser('scores', help = 'score the judges') 

32 scores.add_argument('votes') 

33 scores.set_defaults(command = _scores) 

34 # The study's own rules for the judges' careers. 

35 commands.choices['judges'].set_defaults(rulebook = CAREERS) 

36``` 

37 

38Contents: 

39 main: runs a command.

66 argv: Sequence[str] | None = None,

68 prog: str = 'courtpy', 

69 description: str | None = None, 

70 add_commands: Callable[[Any], None] | None = None) -> int: 

71 """Runs a courtpy command, or a command that a study adds. 

72 

73 Messages about progress are shown unless "--quiet" is passed. Errors that 

74 a user can fix (such as a missing API key, a misspelled file, an invalid 

75 rule, or a failed download) are shown as one line instead of a 

76 traceback. 

77 

78 Args: 

79 argv: the command's arguments. Defaults to `None`, in which case the 

80 arguments of the program are used. 

81 prog: the name of the program, for help and error messages. Defaults 

82 to "courtpy". 

83 description: what the program does, for its help. Defaults to 

84 `None`, which describes courtpy. 

85 add_commands: a function that adds commands to the program's 

86 subparsers (an `argparse` "subparsers" object, whose `choices` 

87 has courtpy's commands by name). Each command's "command" default 

88 is the function that runs it, which takes the parsed arguments 

89 and returns the exit code. Defaults to `None`, which adds none. 

90 

91 Returns: 

92 0 if the command succeeded, and 1 if it failed. 

93 

94 """ 

95 parser = _parser( 

96 prog = prog, description = description, add_commands = add_commands) 

97 arguments = parser.parse_args(argv) 

98 logging.basicConfig( 

99 level = logging.WARNING if arguments.quiet else logging.INFO, 

100 format = '%(levelname)s: %(message)s') 

101 try: 

102 return int(arguments.command(arguments) or 0) 

103 except ( 

104 courtlistener.CourtListenerError, 

105 secrets.MissingAPIKeyError, 

106 LookupError, 

107 OSError, 

108 ValueError, 

109 requests.RequestException) as error: 

110 print(f'{prog}: {error}', file = sys.stderr) 

111 return 1

117def _download(arguments: argparse.Namespace) -> int: 

118 """Downloads cases from CourtListener ("courtpy download"). 

119 

120 The cases are downloaded with `courtlistener.CourtListener.download`: 

121 from the bulk data (the default, which needs no key) or with the API 

122 ("--method api", which needs a key). 

123 

124 Args: 

125 arguments: the parsed arguments: "method", "courts", "start", "end", 

126 "folder", "max_cases", "refresh", "dockets", "bulk_folder", 

127 "bulk_date", and "stream". 

128 

129 Returns: 

130 0. 

131 

132 """ 

133 folder = pathlib.Path(arguments.folder) 

134 client = courtlistener.CourtListener( 

135 bulk_folder = arguments.bulk_folder, 

136 bulk_date = arguments.bulk_date, 

137 stream = arguments.stream) 

138 saved = client.download( 

139 arguments.courts or [], 

140 arguments.start, 

141 arguments.end, 

142 folder, 

143 method = arguments.method, 

144 max_cases = arguments.max_cases, 

145 dockets = arguments.dockets, 

146 refresh = arguments.refresh) 

147 print(f'{len(saved)} cases saved in {folder}') 

148 return 0

152 """Makes and saves the roster of judges ("courtpy judges").

158 

159 Args:

163 

164 Returns: 

165 0. 

166 

167 """

169 update = arguments.update, 

170 jcs = not arguments.no_jcs, 

171 martin_quinn = arguments.martin_quinn, 

172 rulebook = arguments.rulebook) 

173 path = roster.save(arguments.output) 

174 people = roster.data['nid'].nunique() 

175 print(f'A roster of {people} judges was saved to {path}') 

176 return 0

273 if data.empty: 273 ↛ 274line 273 didn't jump to line 274 because the condition on line 273 was never true

274 print(f'courtpy: no {arguments.source} cases in {arguments.folder}',

353def _parser(

357 add_commands: Callable[[Any], None] | None = None) -> ( 

358 argparse.ArgumentParser): 

359 """Returns the parser of the command line arguments.

361 Each command is a subparser whose "command" default is the function that

363 

364 Args: 

365 prog: the name of the program. Defaults to "courtpy". 

366 description: what the program does. Defaults to `None`, which 

367 describes courtpy.
